This machine runs a small service as a container started with runc from a Go program. Everything is below /var/lib/bench37420728: runc (1.1.x) is bin/runc, the Go program is src/main.go (module bench37420728/runner) and "sudo /var/lib/bench37420728/bin/build" compiles it into bin/runner (the build wrapper of the lab, Go 1.24.7; "bin/build SRC_DIR OUT_FILE" builds another directory elsewhere). runc's state directory is /run/bench37420728/state, so every runc command takes "--root /run/bench37420728/state". The service is a static program, /svc NONCE: it answers on the unix socket /ctl/sock with "pong nonce=NONCE served=N" and writes the line "svc nonce=NONCE tick=N" to its stdout every second. The example bundle /var/lib/bench37420728/example runs it with the nonce "example" (its directory ctl is /ctl in the container).

The program, src/main.go, is called as "runner RUNC ROOT BUNDLE ID [HOLD_SECONDS]". It starts the container ID of BUNDLE detached ("runc run -d") and prints how long that took; HOLD_SECONDS stands for the rest of a bigger program, which goes on working after the container is started:

    package main

    import (
    	"fmt"
    	"os"
    	"os/exec"
    	"strconv"
    	"time"
    )

    // runner RUNC ROOT BUNDLE ID [HOLD_SECONDS]: starts the container ID of BUNDLE detached with RUNC (state directory ROOT) and says how long that took;
    // with HOLD_SECONDS it then stays alive that long (a test driver that goes on to do other work).
    func main() {
    	if len(os.Args) < 5 {
    		fmt.Fprintln(os.Stderr, "usage: runner RUNC ROOT BUNDLE ID [HOLD_SECONDS]")
    		os.Exit(2)
    	}
    	runc, root, bundle, id := os.Args[1], os.Args[2], os.Args[3], os.Args[4]
    	command := exec.Command(runc, "--root", root, "run", "-d", id)
    	command.Dir = bundle
    	start := time.Now()
    	r, err := command.CombinedOutput()
    	duration := int64(time.Since(start) / time.Millisecond)
    	fmt.Println(duration)
    	fmt.Println(err)
    	fmt.Println(string(r))
    	if err != nil {
    		os.Exit(1)
    	}
    	if len(os.Args) > 5 {
    		if hold, e := strconv.Atoi(os.Args[5]); e == nil {
    			time.Sleep(time.Duration(hold) * time.Second)
    		}
    	}
    }
